

@commands.command(help="Add a purple overlay on the users avatar", usage="<user>")
async def purple(self, ctx, User: discord.User):
    Embed = bot.DefaultEmbed().set_image(url=f"https://api.no-api-key.com/api/v2/purplify?image={User.avatar_url_as(format='png')}")
    await bot.Helper.SendMessage(ctx, embed=Embed)

@commands.command(help="Make a crap meme", usage="<user>")
async def crap(self, ctx, User: discord.User):
    Image = bot.Helper.BytesToFile(requests.get(f"https://api.no-api-key.com/api/v2/crap?stepped={User.avatar_url_as(format='png')}").content, "image.png")
    Embed = bot.DefaultEmbed().set_image(url="attachment://image.png")
    await bot.Helper.SendMessage(ctx, embed=Embed, file=Image)
